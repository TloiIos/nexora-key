import requests
import time

FIREBASE_URL = "https://keyb-2f31d-default-rtdb.asia-southeast1.firebasedatabase.app/keys"

def validate_key(key_id: str, hwid: str) -> dict:
    """Validate a NEXORA API Key against Firebase Realtime DB."""
    try:
        response = requests.get(f"{FIREBASE_URL}/{key_id}.json", timeout=5)
        response.raise_for_status()
        data = response.json()

        if not data:
            return {"valid": False, "reason": "Key not found"}

        if data.get("status") != "active":
            return {"valid": False, "reason": f"Key status: {data.get('status')}"}

        expires_at = data.get("expiresAt", 0)
        if expires_at and int(time.time() * 1000) > expires_at:
            return {"valid": False, "reason": "Key expired"}

        max_devices = data.get("maxDevices", 1)
        devices = data.get("devices", {})

        if hwid in [d.get("uid") for d in devices.values() if isinstance(d, dict)]:
            return {"valid": True, "reason": "HWID already registered"}

        if len(devices) >= max_devices:
            return {"valid": False, "reason": "Device limit reached"}

        return {"valid": True, "reason": "Key validated successfully"}

    except requests.RequestException as e:
        return {"valid": False, "reason": f"Network error: {e}"}

const https = require('https');

const FIREBASE_URL =
  'https://keyb-2f31d-default-rtdb.asia-southeast1.firebasedatabase.app/keys';

async function validateKey(keyId, hwid) {
  try {
    const res = await fetch(`${FIREBASE_URL}/${keyId}.json`);
    if (!res.ok) return { valid: false, reason: 'Firebase request failed' };

    const data = await res.json();
    if (!data) return { valid: false, reason: 'Key not found in database' };

    if (data.status !== 'active') {
      return { valid: false, reason: `Key status: ${data.status}` };
    }

    if (data.expiresAt && Date.now() > data.expiresAt) {
      return { valid: false, reason: 'Key has expired' };
    }

    const devices = data.devices ? Object.values(data.devices) : [];
    const maxDevices = data.maxDevices || 1;

    if (devices.some(d => d.uid === hwid)) {
      return { valid: true, reason: 'HWID already registered' };
    }

    if (devices.length >= maxDevices) {
      return { valid: false, reason: 'Device limit exceeded' };
    }

    return { valid: true, reason: 'Key validated', packageId: data.packageId };
  } catch (err) {
    return { valid: false, reason: err.message };
  }
}

#include <iostream>
#include <string>
#include <curl/curl.h>
#include <nlohmann/json.hpp>

using json = nlohmann::json;

static size_t WriteCallback(void* contents, size_t size, size_t nmemb, std::string* s) {
    s->append((char*)contents, size * nmemb);
    return size * nmemb;
}

bool ValidateNexoraKey(const std::string& keyId, const std::string& hwid) {
    CURL* curl = curl_easy_init();
    if (!curl) return false;

    std::string url =
        "https://keyb-2f31d-default-rtdb.asia-southeast1.firebasedatabase.app/keys/"
        + keyId + ".json";

    std::string response;
    curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, WriteCallback);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response);
    curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 1L);

    CURLcode res = curl_easy_perform(curl);
    curl_easy_cleanup(curl);

    if (res != CURLE_OK) return false;

    try {
        auto data = json::parse(response);
        if (data.is_null()) return false;
        if (data["status"] != "active") return false;

        auto expiresAt = data["expiresAt"].get<long long>();
        auto now = std::chrono::system_clock::now().time_since_epoch();
        auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(now).count();

        if (ms > expiresAt) return false;
        return true;
    } catch (...) {
        return false;
    }
}

FAQ

Câu Hỏi Thường Gặp

Mua Key xong thì kích hoạt như thế nào?
Sau khi thanh toán qua Zalo, Admin sẽ cấp 1 mã Key (VD: 8Ttza-yTK9p-Qwd0g-QOPPW). Nhập mã này vào Tool/Phần mềm — hệ thống tự động lấy HWID máy và đăng ký trên Firebase.
Đổi máy tính mới thì có dùng lại Key cũ được không?
Hoàn toàn được. Liên hệ Admin qua Zalo — Admin vào Dashboard chọn Tab Thiết bị, bấm "Gỡ HWID" để giải phóng slot cho máy mới.
Một Key có thể sử dụng đồng thời nhiều máy không?
Có, phụ thuộc vào gói License (VD: Starter 1 máy, Pro 2 máy, Developer 5 máy, Lifetime 10 máy). Mỗi máy được đăng ký bằng HWID duy nhất.
Key hết hạn rồi có gia hạn được không?
Được. Liên hệ Admin để gia hạn Key hiện tại hoặc cấp mã Key mới với thời hạn tương ứng. Dữ liệu HWID cũ được giữ nguyên.
